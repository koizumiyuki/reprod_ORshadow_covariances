\section*{Orbital-shadow covariance coefficients}

\paragraph{$k=1$, class V, $(1,2;1,3)$}

$L_+=\{3\}$, $L_-=\{2\}$, $L_0=\{1\}$.

\begin{align*}
  A_{\emptyset} &= \frac{\left(N + 1\right) \left(N - \eta + 1\right)}{N \left(N + 2\right)} \\
  A_{\{1\}} &= - \frac{\left(N + 1\right) \left(\eta + 1\right)}{N \left(N + 2\right)} 
\end{align*}
\begin{align*}
  \Gamma_{\emptyset} &= \frac{\left(N + 1\right) \left(N - \eta + 1\right)}{N \left(N + 2\right)} \\
  \Gamma_{\{1\}} &= - \frac{N + 1}{N} 
\end{align*}

\paragraph{$k=2$, class VI''', $(12,13;12,34)$}

$L_+=\{4\}$, $L_-=\{1\}$, $L_0=\{2,3\}$.

\begin{align*}
  A_{\emptyset} &= - \frac{\left(\eta - 1\right) \left(N - \eta + 1\right) \left(2 N - \eta + 1\right)}{4 \left(N - 1\right) \left(N + 2\right)} \\
  A_{\{2\}} &= - \frac{\left(\eta + 1\right) \left(N - \eta + 1\right)^{2}}{4 \left(N - 1\right) \left(N + 2\right)} \\
  A_{\{3\}} &= - \frac{\left(\eta + 1\right) \left(N - \eta + 1\right) \left(2 N - \eta + 1\right)}{4 \left(N - 1\right) \left(N + 2\right)} \\
  A_{\{2,3\}} &= - \frac{\left(\eta + 1\right) \left(N - \eta - 1\right) \left(N - \eta + 1\right)}{4 \left(N - 1\right) \left(N + 2\right)} 
\end{align*}
\begin{align*}
  \Gamma_{\emptyset} &= - \frac{\left(\eta - 1\right) \left(N - \eta + 1\right) \left(2 N - \eta + 1\right)}{4 \left(N - 1\right) \left(N + 2\right)} \\
  \Gamma_{\{2\}} &= \frac{\left(N - \eta + 1\right) \left(N \eta - 3 N + 2 \eta - 2\right)}{4 \left(N - 1\right) \left(N + 2\right)} \\
  \Gamma_{\{3\}} &= - \frac{\left(N - \eta + 1\right) \left(2 N - \eta + 1\right)}{2 \left(N - 1\right) \left(N + 2\right)} \\
  \Gamma_{\{2,3\}} &= \frac{\left(N + 1\right) \left(N - \eta + 1\right)}{\left(N - 1\right) \left(N + 2\right)} 
\end{align*}

\paragraph{mixed $(1,2)$, class C5, $(1;1;12,13)$}

$L_+=\{3\}$, $L_-=\{2\}$, $L_0=\{1\}$.

\begin{align*}
  A_{\emptyset} &= \frac{\left(N + 1\right) \left(\eta - 1\right) \left(N - \eta + 1\right)}{2 N \left(N + 2\right)} \\
  A_{\{1\}} &= \frac{\left(N - \eta + 1\right) \left(N \eta + 3 N + \eta + 1\right)}{2 N \left(N + 2\right)} 
\end{align*}
\begin{align*}
  \Gamma_{\emptyset} &= \frac{\left(N + 1\right) \left(\eta - 1\right) \left(N - \eta + 1\right)}{2 N \left(N + 2\right)} \\
  \Gamma_{\{1\}} &= \frac{\left(2 N + 1\right) \left(N - \eta + 1\right)}{N \left(N + 2\right)} 
\end{align*}
